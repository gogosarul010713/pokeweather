## Criterios de Aceptación

- [ ] Cada elemento reutiliza la estructura de tarjeta de Clima como base visual (mismo contenedor, border-radius ~10px, alto fijo ~64px)
- [ ] **Layout:** `flex row`, 3 columnas, alineación vertical centrada, padding horizontal 12px, padding vertical 10px
- [ ] **Columna izquierda:** sprite del Pokémon 44×44px, fondo transparente, sin border-radius
- [ ] **Columna central (`flex: 1`):**
  - Línea 1: nombre del lugar — 14px, font-weight 600, color primario
  - Línea 2: ciudad — 12px, color secundario gris
  - Línea 3: país — 12px, color secundario gris
  - Línea 4: badge **Verified** — ícono checkmark en círculo azul + texto, 11px (si aplica)
- [ ] **Columna derecha (`align-items: flex-end`):**
  - Línea 1: pills de tipo con íconos hexagonales del Pokémon
  - Línea 2: hora local del parque — 11px, color terciario gris claro
- [ ] Borde de la tarjeta: color del tipo principal del Pokémon, 0.5px solid
- [ ] Click en tarjeta → abre popup de nido o panel detalle (delegado a siguiente US)
- [ ] Scroll fluido en lista de nidos

## Código de Ejemplo

### NestCard.tsx

```tsx
interface NestCardProps {
  nest: Nest
  isSelected: boolean
  onSelect: (nest: Nest) => void
}

export default function NestCard({ nest, isSelected, onSelect }: NestCardProps) {
  const TYPE_COLORS: Record<string, string> = { /* ... */ }
  const pokemonImageUrl = `/pokemon/${nest.pokemon.toLowerCase()}.png`

  return (
    <div 
      className={`nc-root ${isSelected ? 'selected' : ''}`}
      style={{ borderLeftColor: TYPE_COLORS[nest.pokemonType[0]] }}
      onClick={() => onSelect(nest)}
    >
      {/* Columna izquierda: sprite */}
      <img 
        src={pokemonImageUrl} 
        alt={nest.pokemon}
        className="nc-sprite"
        onError={e => { e.currentTarget.style.display = 'none' }}
      />

      {/* Columna central: info */}
      <div className="nc-body">
        <div className="nc-title">{nest.name}</div>
        <div className="nc-city">{nest.city}</div>
        <div className="nc-country">{nest.country}</div>
        {nest.lastReported && (
          <div className="nc-badge">✅ Verificado</div>
        )}
      </div>

      {/* Columna derecha: tipos + hora */}
      <div className="nc-right">
        <div className="nc-types">
          {nest.pokemonType.map(type => (
            <span 
              key={type} 
              className="nc-type-pill"
              style={{ background: TYPE_COLORS[type] }}
              title={type}
            >
              {type.slice(0, 3).toUpperCase()}
            </span>
          ))}
        </div>
        <div className="nc-local-time">
          {new Date(nest.lastReported || new Date()).toLocaleTimeString()}
        </div>
      </div>
    </div>
  )
}
```

export default function NestFeed({ nests }: NestFeedProps) {
  const selectedNest = useStore(s => s.selectedNest)
  const setSelectedNest = useStore(s => s.setSelectedNest)

  // Aplicar filtros según Zustand state
  const filtered = nests // TODO: aplicar nestFilters

  return (
    <div className="nest-feed">
      {filtered.map(nest => (
        <NestCard
          key={nest.id}
          nest={nest}
          isSelected={selectedNest?.id === nest.id}

  )
